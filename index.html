아래 SQL을 Supabase SQL Editor에서 실행하세요.
create table if not exists hr_tasks (
  id          bigserial primary key,
  title       text        not null,
  member      text        not null,
  cat         text        not null default '기타',
  date        date        not null,
  time        text        default '',
  status      text        not null default 'todo',
  note        text        default '',
  created_at  timestamptz default now()
);
alter table hr_tasks disable row level security;
grant all on table hr_tasks to anon;
grant usage, select on sequence hr_tasks_id_seq to anon;

-- 기존 테이블에 time 컬럼 추가 (이미 테이블이 있는 경우)
alter table hr_tasks add column if not exists time text default '';
